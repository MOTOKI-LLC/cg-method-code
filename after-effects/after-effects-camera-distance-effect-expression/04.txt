var axis = "xRotation"; // X: xRotation / Y: yRotation / Z: zRotation
var cc = effect("targetControl")(1).value;
var result = value;
if (isFinite(cc)) {
    cc = Math.max(0, Math.min(3, cc));
    var n = Math.min(2, Math.floor(cc));
    var a = effect("Target-" + n)(1).transform[axis].value;
    var b = effect("Target-" + (n + 1))(1).transform[axis].value;
    result = effect("ease")(1).value == 1 ?
        ease(cc, n, n + 1, a, b) : linear(cc, n, n + 1, a, b);
}
result;
