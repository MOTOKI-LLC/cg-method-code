/* Import PSD as a composition; move only new direct footage sources. */
(function () {
    var project = app.project;
    if (!project) { alert("プロジェクトを開いてください。"); return; }
    var file = File.openDialog("レイヤーを持つPSDを1つ選択してください");
    if (!file) return;
    if (!file.exists || !/\.psd$/i.test(file.name)) {
        alert("存在する.psdファイルを選択してください。"); return;
    }
    var before = {}, options, imported, moved = 0;
    try {
        options = new ImportOptions(file);
        if (!options.canImportAs(ImportAsType.COMP)) {
            alert("このPSDをコンポジションとして読み込めません。標準の読み込みで確認してください。"); return;
        }
        options.importAs = ImportAsType.COMP;
        options.sequence = false;
        for (var i = 1; i <= project.numItems; i++) before["id_" + project.item(i).id] = true;
    } catch (error) { alert("読み込み前の確認に失敗しました。\n" + error.toString()); return; }
    app.beginUndoGroup("PSDをコンポジションとして読み込み");
    try {
        imported = project.importFile(options);
        if (!(imported instanceof CompItem) || before["id_" + imported.id]) {
            throw new Error("新規コンポジションを取得できませんでした。");
        }
        // Move only newly imported direct sources, identified by Item.id.
        // Retain importer-created folders; never remove a project item.
        var sources = [], seen = {};
        for (var n = 1; n <= imported.numLayers; n++) {
            var source = imported.layer(n).source;
            if (!(source instanceof FootageItem)) continue;
            var key = "id_" + source.id;
            if (before[key] || seen[key]) continue;
            seen[key] = true; sources.push(source);
        }
        for (var j = 0; j < sources.length; j++) {
            sources[j].parentFolder = imported.parentFolder;
            moved++;
        }
        alert("コンポジション「" + imported.name + "」を読み込み、新規素材" + moved + "個を同じ階層へ移しました。空のフォルダは残します。");
    } catch (error) {
        alert("処理が途中で止まりました。一部の読み込み・移動が残る場合があります。編集メニューで取り消して確認してください。\n" + error.toString());
    } finally { app.endUndoGroup(); }
}());
