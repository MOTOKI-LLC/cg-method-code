(function () {
    if (!app.project) { alert("Open a saved practice project first."); return; }
    var selected = app.project.selection.slice(0), plan = [], report = [];
    for (var i = 0; i < selected.length; i++) {
        var item = selected[i];
        if (!(item instanceof FootageItem) || !(item.mainSource instanceof FileSource)) { continue; }
        var extension = item.file.name.match(/\.([A-Za-z0-9]+)$/);
        var name = "cgm_asset_" + item.id + (extension ? "." + extension[1].toLowerCase() : "");
        for (var k = 1; k <= app.project.numItems; k++) {
            var other = app.project.item(k);
            if (other.id != item.id && other.name == name) {
                alert("Name collision: " + name + ". Nothing renamed."); return;
            }
        }
        if (item.name != name) { plan.push({item:item, before:item.name, after:name}); }
    }
    if (!plan.length) { alert("No file-based footage needs renaming."); return; }
    app.beginUndoGroup("Rename selected footage display names");
    try {
        for (var j = 0; j < plan.length; j++) {
            var row = plan[j];
            row.item.name = row.after;
            report.push(row.before + " -> " + row.after);
        }
        alert(report.join("\n") + "\nOnly project display names changed. Source files unchanged.");
    } catch (e) { alert(e.toString() + "\nUndo this group or reopen the saved practice copy."); }
    finally { app.endUndoGroup(); }
}());
