var prefix = "BGM_";
var audioOnly = true; // falseならIN/OUTだけで判定
var result = "";
for (var i = thisLayer.index + 1; i <= thisComp.numLayers; i++) {
    var layer = thisComp.layer(i);
    if (layer.name.indexOf(prefix) == 0 && layer.inPoint <= time && time < layer.outPoint &&
            (!audioOnly || layer.audioActive)) {
        result = layer.name;
        break;
    }
}
result;
