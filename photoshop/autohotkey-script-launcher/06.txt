; Photoshop用スクリプトランチャー

#IfWinActive ahk_class Photoshop
; Alt二連打で起動

PutEsc:
  Send,{alt}
return

$alt::
  WinGet, AWinHwnd, ID, A
  if (BefAWinHwnd == AWinHwnd) && (A_PriorHotkey == A_ThisHotkey) && (500 > A_TimeSincePriorHotkey) {

Gui,+AlwaysOnTop
Gui, Add, Tab3,, General|View|Settings

Gui, 1:Add, Button,W200 Gscript1, アラート1
Gui, 1:Add, Button,W200 Gscript2, アラート2
Gui, 1:Add, Button,W200 Gscript3, アラート3

Gui, Tab, 2
Gui, 1:Add, Button,W200 Gscript4, アラート4
Gui, 1:Add, Button,W200 Gscript5, アラート5
Gui, 1:Add, Button,W200 Gscript6, アラート6

Gui, Tab, 3
Gui, 1:Add, Button,W200 Gscript4, アラート7
Gui, 1:Add, Button,W200 Gscript5, アラート8
Gui, 1:Add, Button,W200 Gscript6, アラート9

Gui,1:Show,,スクリプトランチャー
return

  script1:
  Gui,1:
  #IfWinActive ahk_class Photoshop
  app:=ComObjCreate("Photoshop.Application")
  app.DoJavaScriptFile("sample.jsx")
  return


  } else {
    BefAWinHwnd := AWinHwnd
    SetTimer, PutEsc, -500
  }
  keywait, alt
return
