using UnityEngine;

[RequireComponent(typeof(AudioSource))]
public class PlaySound : MonoBehaviour
{
    public AudioClip sound01;
    public AudioClip sound02;
    private AudioSource audioSource;

    void Awake()
    {
        audioSource = GetComponent<AudioSource>();
        if (sound01 == null || sound02 == null)
            Debug.LogWarning("Sound 01 / Sound 02にAudioClipを設定してください。", this);
    }

    void Update()
    {
        if (audioSource == null || !audioSource.isActiveAndEnabled) return;
        if (Input.GetMouseButtonUp(0) && sound01 != null)
            audioSource.PlayOneShot(sound01);
        if (Input.GetMouseButtonUp(1) && sound02 != null)
            audioSource.PlayOneShot(sound02);
    }
}
