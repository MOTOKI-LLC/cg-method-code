import bpy

ob = bpy.context.object
if ob is None or ob.type != "MESH" or ob.data.shape_keys is None:
    raise RuntimeError("シェイプキーのある練習用メッシュをオブジェクトモードで選んでください。")
if ob.mode != "OBJECT" or not ob.data.shape_keys.use_relative:
    raise RuntimeError("オブジェクトモード・相対シェイプキーで実行してください。")
mixed = ob.shape_key_add(name="SavedMix", from_mix=True)
for key in ob.data.shape_keys.key_blocks:
    key.value = 0.0
mixed.value = 1.0
