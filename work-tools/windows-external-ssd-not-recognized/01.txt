$ErrorActionPreference = 'Stop'
Get-PnpDevice -PresentOnly -Class DiskDrive |
    Select-Object Status,Class,FriendlyName | Format-Table -AutoSize
Get-Disk |
    Select-Object Number,BusType,OperationalStatus,PartitionStyle,
        @{n='GB';e={[math]::Round($_.Size/1GB)}} | Format-Table -AutoSize
Get-Partition |
    Select-Object DiskNumber,PartitionNumber,DriveLetter,Type,
        @{n='GB';e={[math]::Round($_.Size/1GB)}} | Format-Table -AutoSize
Get-Volume |
    Select-Object DriveLetter,FileSystem,HealthStatus,
        @{n='GB';e={[math]::Round($_.Size/1GB)}} | Format-Table -AutoSize
