"""Convert explicit UNC/SMB strings only. No network, mount lookup or clipboard."""
import argparse
import re
from urllib.parse import quote, unquote, urlsplit


def host_ok(host):
    if not re.fullmatch(r"[A-Za-z0-9](?:[A-Za-z0-9.-]*[A-Za-z0-9])?", host):
        raise ValueError("Use a plain server hostname; credentials/ports are unsupported.")
    return host


def segments_ok(parts):
    if not parts or any(not p or p in (".", "..") or
                        any(c in p for c in '\\/:*?"<>|') or
                        any(ord(c) < 32 for c in p) or p.endswith((" ", "."))
                        for p in parts):
        raise ValueError("Invalid share/path component.")
    return parts


def unc_to_smb(value):
    if not value.startswith("\\\\"):
        raise ValueError("Expected \\\\server\\share path, not a local drive.")
    host, *parts = value[2:].split("\\")
    parts = segments_ok(parts)
    return "smb://" + host_ok(host) + "/" + "/".join(quote(p, safe="") for p in parts)


def smb_to_unc(value):
    if any(ord(c) < 32 for c in value):
        raise ValueError("Control characters are unsupported.")
    u = urlsplit(value)
    if u.scheme != "smb" or u.query or u.fragment or not u.netloc:
        raise ValueError("Expected smb://server/share URL without query/fragment.")
    host = host_ok(u.netloc)
    encoded = u.path[1:].split("/") if u.path.startswith("/") else []
    if any(re.search(r"%(?![0-9A-Fa-f]{2})", p) for p in encoded):
        raise ValueError("Malformed percent escape.")
    parts = segments_ok([unquote(p, encoding="utf-8", errors="strict") for p in encoded])
    return "\\\\" + host + "\\" + "\\".join(parts)


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("direction", choices=["unc-to-smb", "smb-to-unc"])
    parser.add_argument("paths", nargs="+")
    args = parser.parse_args()
    convert = unc_to_smb if args.direction == "unc-to-smb" else smb_to_unc
    try:
        results = [convert(p) for p in args.paths]
    except (ValueError, UnicodeError) as error:
        parser.error(str(error))
    print("\n".join(results))
