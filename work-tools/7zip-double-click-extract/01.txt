param(
    [Parameter(Mandatory=$true)][string]$SevenZip,
    [Parameter(Mandatory=$true)][string]$Archive,
    [Parameter(Mandatory=$true)][string]$OutputFolder
)
$ErrorActionPreference = 'Stop'
if (-not (Test-Path -LiteralPath $SevenZip -PathType Leaf) -or
    -not (Test-Path -LiteralPath $Archive -PathType Leaf)) {
    throw '7-Zip or archive does not exist; extraction was not started.'
}
if (Test-Path -LiteralPath $OutputFolder) {
    throw 'Output already exists; choose a new folder.'
}
& $SevenZip 't' $Archive
if ($LASTEXITCODE -ne 0) { throw 'Archive test failed; extraction was not started.' }
& $SevenZip 'x' $Archive ('-o' + $OutputFolder) '-aos'
if ($LASTEXITCODE -ne 0) { throw 'Extraction failed; inspect the new output folder.' }
